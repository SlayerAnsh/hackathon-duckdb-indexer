WITH t AS (
  SELECT value AS type, count(*) AS n,
         row_number() OVER (ORDER BY count(*) DESC) AS rk
  FROM attributes
  WHERE name = 'type'
  GROUP BY value
)
SELECT CASE WHEN rk <= 7 THEN type ELSE 'Other' END AS type,
       sum(n) AS entities
FROM t
GROUP BY 1
ORDER BY entities DESC;

SQL 
SELECT CASE
         WHEN e.expires_at - h.block < 1800   THEN '< 1 h'
         WHEN e.expires_at - h.block < 10800  THEN '1 to 6 h'
         WHEN e.expires_at - h.block < 43200  THEN '6 to 24 h'
         WHEN e.expires_at - h.block < 302400 THEN '1 to 7 d'
         ELSE '> 7 d' END  AS bucket,
       a.value              AS type,
       count(*)             AS entities
FROM entities e
JOIN attributes a ON a.key = e.key AND a.name = 'type'
CROSS JOIN live_head h
GROUP BY ALL;

SQL 
SELECT rank() OVER (ORDER BY count(*) DESC) AS rank,
       e.creator,
       count(*)                AS entities,
       count(DISTINCT a.value) AS types,
       sum(count(*)) OVER (ORDER BY count(*) DESC)
         / sum(count(*)) OVER ()  AS running_share
FROM entities e
JOIN attributes a ON a.key = e.key AND a.name = 'type'
GROUP BY e.creator
ORDER BY rank;

-- Open tickets with the most discussion, highest priority first
SELECT t.key,
       ti.value          AS title,
       p.num             AS priority,
       count(c.key)      AS comments,
       max(c.updated_at) AS last_comment_block
FROM entities t
JOIN attributes ty ON ty.key = t.key AND ty.name = 'type'   AND ty.value = 'ticket'
JOIN attributes st ON st.key = t.key AND st.name = 'status' AND st.value = 'open'
JOIN attributes ti ON ti.key = t.key AND ti.name = 'title'
JOIN attributes p  ON p.key  = t.key AND p.name  = 'priority'
LEFT JOIN attributes ref ON ref.name = 'ticket' AND ref.value = t.key
LEFT JOIN entities c     ON c.key = ref.key
GROUP BY t.key, ti.value, p.num
ORDER BY p.num ASC, comments DESC;
